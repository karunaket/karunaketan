So I spent over a month of focused solo work turning JumpRift: 3D Platformer into a full public mobile release: reworking camera framing for phones, tuning touch controls, building Night Mode, fixing mobile frame drops, and handling the Google Play Console pipeline myself.

03. Game Feel & Mobile Controls

  • Forgiving Jump Windows: Added subtle coyote time and input buffering so tapping jump a split second early or right after leaving a ledge still registers.
  • Under-0.5s Respawns: Hitting a checkpoint saves progress right away, and falling drops you straight back in with zero UI popups.
  • Thumb-Arc Ergonomics: Placed the virtual stick and jump button inside natural thumb reach to prevent accidental camera drags mid-jump.
  • Steady Mobile Frame Pacing: Kept lighting lightweight so mid-range Android phones hold smooth FPS in both Day and Night modes.

Zone 02 // Unreal Engine 5 Blueprints & Gameplay Tools

Technical Design & Gameplay Systems

07. Technical Design: Designer Tools & Android Services

To make level tuning fast inside the Unreal Editor and connect checkpoints to Google Play achievements, I set up two small helper classes (AMovingPlatform and UJumpRiftGameInstance) that expose clean sliders and Blueprint nodes for level triggers.

08. Designer-Tunable Platform Actor (AMovingPlatform)

Instead of duplicating Timeline nodes across dozens of bridges and sweepers, I used one base actor (AMovingPlatform) inherited by Blueprint variants (BP_MovingPlatform, BP_RotatingBlocker, BP_UpDownBlocker):

  • Viewport Tuning: Exposed PlatformVelocity, MoveDistance, and RotationVelocity in the Details panel so obstacle rhythm can be tuned live while testing jumps.
  • Local vs. World Space Toggle (bUseLocalSpaceMovement): Moves angled platforms along their local Forward/Right/Up axes so tilted ramps oscillate cleanly along their slope.
  • Zero-Drift Endpoint Reversal: Checks distance from StartLocation and TargetLocation so platforms never drift out of alignment over time on phones.
BP_MovingPlatform inheriting AMovingPlatform with exposed velocity, distance, and local-space toggles in the Details panel.
MovingPlatform.h 
#pragma once

#include "CoreMinimal.h"
#include "GameFramework/Actor.h"
#include "MovingPlatform.generated.h"

UCLASS()
class OBSTACLEASSAULT_API AMovingPlatform : public AActor
{
	GENERATED_BODY()

public:
	AMovingPlatform();
	virtual void Tick(float DeltaTime) override;

protected:
	virtual void BeginPlay() override;

private:
	UPROPERTY(EditAnywhere, Category = "Moving")
	FVector PlatformVelocity = FVector(100, 0, 0);

	UPROPERTY(EditAnywhere, Category = "Moving")
	float MoveDistance = 100;

	UPROPERTY(EditAnywhere, Category = "Rotation")
	FRotator RotationVelocity;

	UPROPERTY(EditAnywhere, Category = "Moving")
	bool bUseLocalSpaceMovement = false;

	FVector StartLocation;
	FVector TargetLocation;

	void MovePlatform(float DeltaTime);
	void RotatePlatform(float DeltaTime);
};
MovingPlatform.cpp 
#include "MovingPlatform.h"

AMovingPlatform::AMovingPlatform()
{
	PrimaryActorTick.bCanEverTick = true;
}

void AMovingPlatform::BeginPlay()
{
	Super::BeginPlay();
	StartLocation = GetActorLocation();

	if (bUseLocalSpaceMovement)
	{
		FVector LocalMoveDirection = PlatformVelocity.GetSafeNormal();
		FVector WorldDirection =
			GetActorForwardVector() * LocalMoveDirection.X +
			GetActorRightVector() * LocalMoveDirection.Y +
			GetActorUpVector() * LocalMoveDirection.Z;

		WorldDirection.Normalize();
		TargetLocation = StartLocation + WorldDirection * MoveDistance;
	}
	else
	{
		TargetLocation = StartLocation + PlatformVelocity.GetSafeNormal() * MoveDistance;
	}
}

void AMovingPlatform::Tick(float DeltaTime)
{
	Super::Tick(DeltaTime);
	MovePlatform(DeltaTime);
	RotatePlatform(DeltaTime);
}

void AMovingPlatform::MovePlatform(float DeltaTime)
{
	FVector CurrentLocation = GetActorLocation();
	FVector CurrentWorldVelocity;

	if (bUseLocalSpaceMovement)
	{
		FVector LocalVelocityDirection = PlatformVelocity.GetSafeNormal();
		CurrentWorldVelocity =
			GetActorForwardVector() * LocalVelocityDirection.X +
			GetActorRightVector() * LocalVelocityDirection.Y +
			GetActorUpVector() * LocalVelocityDirection.Z;

		CurrentWorldVelocity *= PlatformVelocity.Size();
	}
	else
	{
		CurrentWorldVelocity = PlatformVelocity;
	}

	CurrentLocation += CurrentWorldVelocity * DeltaTime;
	SetActorLocation(CurrentLocation);

	float DistanceFromStart = FVector::Dist(StartLocation, CurrentLocation);
	float DistanceFromTarget = FVector::Dist(TargetLocation, CurrentLocation);

	if (DistanceFromStart >= MoveDistance && FVector::DotProduct(CurrentWorldVelocity, (CurrentLocation - StartLocation)) > 0)
	{
		PlatformVelocity = -PlatformVelocity;
	}
	else if (DistanceFromTarget >= MoveDistance && FVector::DotProduct(CurrentWorldVelocity, (CurrentLocation - TargetLocation)) > 0)
	{
		PlatformVelocity = -PlatformVelocity;
	}
}

void AMovingPlatform::RotatePlatform(float DeltaTime)
{
	AddActorLocalRotation(RotationVelocity * DeltaTime);
}

JumpRiftGameInstance.cpp (Cache-Safe Achievement Unlock & Polling) 
bool UJumpRiftGameInstance::RequestGooglePlayAchievementUnlock(APlayerController* LocalPlayerController, const FString& AchievementName)
{
    if (AchievementName.IsEmpty()) return false;

#if !PLATFORM_ANDROID
    return false;
#else
    const UAndroidRuntimeSettings* AndroidSettings = GetDefault<UAndroidRuntimeSettings>();
    const FGooglePlayAchievementMapping* Mapping = AndroidSettings
        ? AndroidSettings->AchievementMap.FindByPredicate([&AchievementName](const FGooglePlayAchievementMapping& Candidate)
            {
                return Candidate.Name.Equals(AchievementName, ESearchCase::CaseSensitive);
            })
        : nullptr;

    if (!Mapping || Mapping->AchievementID.IsEmpty()) return false;
    if (!IsValid(LocalPlayerController) || !LocalPlayerController->IsLocalController()) return false;

    IOnlineSubsystem* OnlineSubsystem = IOnlineSubsystem::Get();
    if (!OnlineSubsystem || OnlineSubsystem->GetSubsystemName() != FName(TEXT("GOOGLEPLAY"))) return false;

    IOnlineIdentityPtr IdentityInterface = OnlineSubsystem->GetIdentityInterface();
    if (!IdentityInterface.IsValid() || IdentityInterface->GetLoginStatus(0) != ELoginStatus::LoggedIn) return false;

    const FUniqueNetIdPtr OSSUserId = IdentityInterface->GetUniquePlayerId(0);
    IOnlineAchievementsPtr Achievements = OnlineSubsystem->GetAchievementsInterface();
    if (!OSSUserId.IsValid() || !Achievements.IsValid()) return false;

    const FString RequestKey = MakeAchievementRequestKey(AchievementName, *OSSUserId);
    if (PendingAchievementUnlocks.Contains(RequestKey) || CompletedAchievementUnlocks.Contains(RequestKey))
    {
        return false;
    }

    FOnlineAchievement CachedAchievement;
    if (Achievements->GetCachedAchievement(*OSSUserId, AchievementName, CachedAchievement) != EOnlineCachedResult::Success)
    {
        FPendingGooglePlayAchievementUnlock QueuedRequest;
        QueuedRequest.AchievementName = AchievementName;
        QueuedRequest.UserId = OSSUserId;
        PendingCacheAchievementUnlocks.Add(RequestKey, MoveTemp(QueuedRequest));
        StartAchievementCachePolling();
        return true;
    }

    return SubmitCachedAchievementUnlock(AchievementName, OSSUserId, CachedAchievement);
#endif
}
